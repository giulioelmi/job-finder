# Autofill Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a profile-based autofill capability to the Bamboo Chrome extension that fills job application forms using DeepSeek for field mapping and optionally generates cover letters.

**Architecture:** Profiles are stored in a new `autofill_profiles` Supabase table and managed on jobbamboo.com. At fill time the extension scans the visible form fields, sends them with the selected profile to a backend endpoint, and DeepSeek returns a selector→value map which the extension applies. Cover letter generation is a separate optional call.

**Tech Stack:** Next.js 16 (App Router, edge runtime for LLM routes), Supabase, DeepSeek API (OpenAI-compatible), Chrome MV3 content script + service worker, Tailwind CSS + shadcn/ui components.

import { NextRequest, NextResponse } from 'next/server'
import { createClient, createServiceClient } from '@/lib/supabase/server'

async function resolveUserId(req: NextRequest): Promise<string | null> {
  const authHeader = req.headers.get('Authorization')
  const bearerToken = authHeader?.startsWith('Bearer ') ? authHeader.slice(7) : null

  if (bearerToken) {
    const service = createServiceClient()
    const { data: { user } } = await service.auth.getUser(bearerToken)
    return user?.id ?? null
  }

  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  return user?.id ?? null
}

export async function GET(req: NextRequest) {
  const userId = await resolveUserId(req)
  if (!userId) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const service = createServiceClient()
  const { data, error } = await service
    .from('autofill_profiles')
    .select('*')
    .eq('user_id', userId)
    .order('created_at', { ascending: true })

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })
  return NextResponse.json({ profiles: data })
}

export async function POST(req: NextRequest) {
  const userId = await resolveUserId(req)
  if (!userId) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const body = await req.json().catch(() => null)
  if (!body?.name?.trim()) {
    return NextResponse.json({ error: 'name is required' }, { status: 400 })
  }

  const service = createServiceClient()
  const { data, error } = await service
    .from('autofill_profiles')
    .insert({ ...body, user_id: userId })
    .select('*')
    .single()

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })
  return NextResponse.json({ profile: data }, { status: 201 })
}

import { NextRequest, NextResponse } from 'next/server'
import { createClient, createServiceClient } from '@/lib/supabase/server'

async function resolveUserId(req: NextRequest): Promise<string | null> {
  const authHeader = req.headers.get('Authorization')
  const bearerToken = authHeader?.startsWith('Bearer ') ? authHeader.slice(7) : null

  if (bearerToken) {

  return user?.id ?? null

export async function PATCH(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const userId = await resolveUserId(req)
  if (!userId) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { id } = await params
  const body = await req.json().catch(() => null)
  if (!body) return NextResponse.json({ error: 'Invalid body' }, { status: 400 })

  // Strip fields the client must not overwrite
  const { id: _id, user_id: _uid, created_at: _ca, ...updates } = body

  const service = createServiceClient()
  const { data, error } = await service
    .from('autofill_profiles')
    .update({ ...updates, updated_at: new Date().toISOString() })
    .eq('id', id)
    .eq('user_id', userId)
    .select('*')
    .single()

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })
  if (!data) return NextResponse.json({ error: 'Not found' }, { status: 404 })
  return NextResponse.json({ profile: data })
}

export async function DELETE(
  req: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  const userId = await resolveUserId(req)
  if (!userId) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const { id } = await params
  const service = createServiceClient()
  const { error } = await service
    .from('autofill_profiles')
    .delete()
    .eq('id', id)
    .eq('user_id', userId)

  if (error) return NextResponse.json({ error: error.message }, { status: 500 })
  return NextResponse.json({ success: true })
}

// src/app/api/extension/autofill/route.ts
import { NextRequest, NextResponse } from 'next/server'
import { createServiceClient } from '@/lib/supabase/server'
import type { AutofillProfile } from '@/lib/types'

export const runtime = 'edge'

const SYSTEM_PROMPT = `You map job application form fields to a user's profile.

Given a JSON array of form fields and a user profile, return STRICT JSON:
{"fields": [{"selector": "...", "value": "..."}]}

Rules:
- Only include fields where the profile has a clear match.
- For select/radio fields the value MUST exactly match one of the provided option strings.
- For checkboxes use "true" or "false".
- For DEI fields where profile value is null, use "Prefer not to say" if that option exists, otherwise skip.
- Never invent values not in the profile.
- Omit fields you are uncertain about.
- Return valid JSON only, no prose, no markdown.`

function serializeProfile(p: AutofillProfile): string {
  const lines: string[] = [
    `Name: ${[p.first_name, p.last_name].filter(Boolean).join(' ')}`,
    `Email: ${p.email ?? ''}`,
    `Phone: ${p.phone ?? ''}`,
    `Address: ${[p.street_address, p.city, p.state, p.zip_code, p.country].filter(Boolean).join(', ')}`,
    `LinkedIn: ${p.linkedin_url ?? ''}`,
    `GitHub: ${p.github_url ?? ''}`,
    `Portfolio: ${p.portfolio_url ?? ''}`,
    `Website: ${p.website_url ?? ''}`,
    `Authorized to work in US: ${p.authorized_to_work === true ? 'Yes' : p.authorized_to_work === false ? 'No' : ''}`,
    `Requires visa sponsorship: ${p.requires_sponsorship === true ? 'Yes' : p.requires_sponsorship === false ? 'No' : ''}`,
    `Visa type: ${p.visa_type ?? ''}`,
    `Desired salary: ${p.desired_salary ? `${p.desired_salary} (${p.salary_type ?? ''})` : ''}`,
    `Willing to relocate: ${p.willing_to_relocate === true ? 'Yes' : p.willing_to_relocate === false ? 'No' : ''}`,
    `Available start date: ${p.available_start_date ?? ''}`,
    `Gender: ${p.gender ?? 'Prefer not to say'}`,
    `Pronouns: ${p.pronouns ?? ''}`,
    `Ethnicity: ${p.ethnicity ?? 'Decline to self identify'}`,
    `Veteran status: ${p.veteran_status ?? "I don't wish to answer"}`,
    `Disability status: ${p.disability_status ?? "I don't wish to answer"}`,
    `Skills: ${(p.skills ?? []).join(', ')}`,
  ]

  if (p.work_experience?.length) {
    lines.push('Work Experience:')
    for (const w of p.work_experience) {
      lines.push(`  - ${w.title} at ${w.company} (${w.start_date} – ${w.current ? 'Present' : w.end_date})`)
      if (w.description) lines.push(`    ${w.description}`)
    }
  }

  if (p.education?.length) {
    lines.push('Education:')
    for (const e of p.education) {
      lines.push(`  - ${e.degree} in ${e.field_of_study} at ${e.school} (${e.start_date} – ${e.end_date})${e.gpa ? `, GPA: ${e.gpa}` : ''}`)
    }
  }

  if (p.projects?.length) {
    lines.push('Projects:')
    for (const pr of p.projects) {
      lines.push(`  - ${pr.name}: ${pr.description} (${pr.technologies})${pr.url ? ` – ${pr.url}` : ''}`)
    }
  }

  if (p.custom_qa?.length) {
    lines.push('Custom Q&A:')
    for (const qa of p.custom_qa) {
      lines.push(`  Q: ${qa.question}`)
      lines.push(`  A: ${qa.answer}`)
    }
  }

  return lines.filter(l => !l.endsWith(': ')).join('\n')
}

export async function POST(req: NextRequest) {
  const authHeader = req.headers.get('Authorization')
  const bearerToken = authHeader?.startsWith('Bearer ') ? authHeader.slice(7) : null
  if (!bearerToken) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const service = createServiceClient()
  const { data: { user } } = await service.auth.getUser(bearerToken)
  if (!user) return NextResponse.json({ error: 'Unauthorized' }, { status: 401 })

  const apiKey = process.env.DEEPSEEK_API_KEY
  if (!apiKey) return NextResponse.json({ error: 'Server misconfigured' }, { status: 500 })

  const body = await req.json().catch(() => null) as {
    profile_id: string
    form_fields: { selector: string; type: string; label: string; placeholder?: string; options?: string[] }[]
    job_title?: string
    company?: string
    profile_overrides?: Partial<AutofillProfile>
  } | null

  if (!body?.profile_id || !body?.form_fields?.length) {
    return NextResponse.json({ error: 'profile_id and form_fields are required' }, { status: 400 })
  }

  const { data: profile } = await service
    .from('autofill_profiles')
    .select('*')
    .eq('id', body.profile_id)
    .eq('user_id', user.id)
    .single()

  if (!profile) return NextResponse.json({ error: 'Profile not found' }, { status: 404 })

  // Merge any inline edits the user made in the extension review card
  const mergedProfile = { ...profile, ...(body.profile_overrides ?? {}) }
  const profileText = serializeProfile(mergedProfile as AutofillProfile)
  const fieldsJson = JSON.stringify(body.form_fields.slice(0, 120))

  const userMessage = [
    body.job_title || body.company
      ? `Job: ${body.job_title ?? ''} at ${body.company ?? ''}`
      : null,
    `Profile:\n${profileText}`,
    `Form fields:\n${fieldsJson}`,
  ].filter(Boolean).join('\n\n')

  try {
    const res = await fetch('https://api.deepseek.com/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${apiKey}`,
      },
      body: JSON.stringify({
        model: 'deepseek-chat',
        temperature: 0,
        max_tokens: 2000,
        response_format: { type: 'json_object' },
        messages: [
          { role: 'system', content: SYSTEM_PROMPT },
          { role: 'user', content: userMessage },
        ],
      }),
    })

    if (!res.ok) {
      const text = await res.text().catch(() => '')
      return NextResponse.json(
        { error: 'DEEPSEEK_ERROR', detail: `${res.status}: ${text.slice(0, 200)}` },
        { status: 502 }
      )
    }

    const data = await res.json()
    const raw = data?.choices?.[0]?.message?.content
    if (!raw) return NextResponse.json({ fields: [] })

    const parsed = JSON.parse(raw) as { fields?: { selector: string; value: string }[] }
    return NextResponse.json({ fields: parsed.fields ?? [] })
  } catch (e) {
    const detail = e instanceof Error ? e.message : String(e)
    return NextResponse.json({ error: 'AUTOFILL_FAILED', detail }, { status: 500 })
  }
}

  const [country, setCountry] = useState(initialProfile?.country ?? '')

    setEducation(prev => prev.map((e, idx) => idx === i ? { ...e, [field]: value } : e))
  }
  function removeEducation(i: number) {
    setEducation(prev => prev.filter((_, idx) => idx !== i))
  }

  function addWorkExp() {
    setWorkExp(prev => [...prev, { company: '', title: '', start_date: '', end_date: '', current: false, description: '' }])

    setWorkExp(prev => prev.map((e, idx) => idx === i ? { ...e, [field]: value } : e))
  }
  function removeWorkExp(i: number) {
    setWorkExp(prev => prev.filter((_, idx) => idx !== i))
  }

  function addProject() {
    setProjects(prev => [...prev, { name: '', url: '', description: '', technologies: '' }])
  }
  function updateProject(i: number, field: keyof ProjectEntry, value: string) {
    setProjects(prev => prev.map((e, idx) => idx === i ? { ...e, [field]: value } : e))
  }
  function removeProject(i: number) {
    setProjects(prev => prev.filter((_, idx) => idx !== i))

      willing_to_relocate: willingToRelocate === 'yes' ? true : willingToRelocate === 'no' ? false : null,
      available_start_date: startDate.trim() || null,
      gender: gender || null,

      ethnicity: ethnicity || null,

      projects,

      cover_letter_prompt: coverLetterPrompt.trim() || null,
    }

      body: JSON.stringify(payload),
    })

    if (res.ok) {

      const data = await res.json().catch(() => ({}))
      setError(data.error ?? 'Failed to save')
    }
  }

      </Card>

        <CardHeader><CardTitle>Personal Info</CardTitle></CardHeader>

            <Button variant="outline" onClick={() => addSkill(skillInput)} disabled={!skillInput.trim()}>Add</Button>
          </div>
        </CardContent>
      </Card>

      <Card>
        <CardHeader>
          <div className="flex items-center justify-between">
            <CardTitle>Custom Q&amp;A</CardTitle>
            <Button variant="outline" size="sm" onClick={addQA}><Plus className="w-4 h-4 mr-1" />Add</Button>
          </div>
          <p className="text-sm text-gray-500 mt-1">Pre-set answers to recurring questions not covered above.</p>
        </CardHeader>
        <CardContent className="space-y-3">
          {customQA.map((qa, i) => (
            <div key={i} className="flex gap-3 items-start">
              <div className="flex-1 grid grid-cols-2 gap-3">
                <input className={inputCls} value={qa.question} onChange={ev => updateQA(i, 'question', ev.target.value)} placeholder="Question" />
                <input className={inputCls} value={qa.answer} onChange={ev => updateQA(i, 'answer', ev.target.value)} placeholder="Answer" />
              </div>
              <button onClick={() => removeQA(i)} className="mt-2 text-gray-400 hover:text-gray-700"><X className="w-4 h-4" /></button>
            </div>
          ))}
          {customQA.length === 0 && <p className="text-sm text-gray-400">No entries yet.</p>}

      </Card>

      {/* Cover letter prompt */}
      <Card>
        <CardHeader>
          <CardTitle>Cover Letter Instructions</CardTitle>
          <p className="text-sm text-gray-500 mt-1">Custom instructions for the AI when generating your cover letter.</p>
        </CardHeader>
        <CardContent>
          <textarea
            className={inputCls + ' min-h-[100px]'}
            value={coverLetterPrompt}
            onChange={e => setCoverLetterPrompt(e.target.value)}
            placeholder="e.g. Write in a confident but concise tone, max 3 paragraphs, emphasize my ML skills and open-source projects."
          />
        </CardContent>
      </Card>

      {error && <p className="text-sm text-red-500">{error}</p>}

      <div className="flex gap-3">
        <Button onClick={handleSave} disabled={saving} size="lg">
          {saving ? 'Saving…' : isEditing ? 'Save changes' : 'Create profile'}
        </Button>
        <Button variant="outline" size="lg" onClick={() => router.push('/profiles')}>Cancel</Button>
      </div>
    </div>
  )
}

```javascript
  // ─── DOM scanner ─────────────────────────────────────────────────────────
  // Scans visible form fields and returns a compact descriptor array for the API.

  function isVisible(el) {
    const rect = el.getBoundingClientRect()
    if (rect.width === 0 && rect.height === 0) return false
    const style = window.getComputedStyle(el)
    return style.display !== 'none' && style.visibility !== 'hidden' && style.opacity !== '0'
  }

  function getFieldSelector(el) {
    if (el.id) return `#${CSS.escape(el.id)}`
    if (el.name) return `[name="${CSS.escape(el.name)}"]`
    const tag = el.tagName.toLowerCase()
    const siblings = Array.from((el.parentElement || document.body).querySelectorAll(tag))
    return `${tag}:nth-of-type(${siblings.indexOf(el) + 1})`
  }

  function getFieldLabel(el) {
    if (el.id) {
      const lbl = document.querySelector(`label[for="${CSS.escape(el.id)}"]`)
      if (lbl) return lbl.textContent.trim().slice(0, 100)
    }
    const ariaLabel = el.getAttribute('aria-label')
    if (ariaLabel) return ariaLabel.trim().slice(0, 100)
    const labelledById = el.getAttribute('aria-labelledby')
    if (labelledById) {
      const lblEl = document.getElementById(labelledById)
      if (lblEl) return lblEl.textContent.trim().slice(0, 100)
    }
    const closestLabel = el.closest('label')
    if (closestLabel) return closestLabel.textContent.replace(el.value || '', '').trim().slice(0, 100)
    if (el.placeholder) return el.placeholder.trim().slice(0, 100)
    const prev = el.previousElementSibling
    if (prev && prev.textContent) return prev.textContent.trim().slice(0, 100)
    return ''
  }

  function getRadioGroupOptions(name) {
    return Array.from(document.querySelectorAll(`input[type="radio"][name="${CSS.escape(name)}"]`))
      .map(r => {
        if (r.id) {
          const lbl = document.querySelector(`label[for="${CSS.escape(r.id)}"]`)
          if (lbl) return lbl.textContent.trim()
        }
        return r.closest('label')?.textContent?.trim() || r.value
      })
      .filter(Boolean)
  }

  function scanFormFields() {
    const fields = []
    const seen = new Set()
    const seenRadioGroups = new Set()

    const els = document.querySelectorAll(
      'input:not([type="hidden"]):not([type="submit"]):not([type="button"])' +
      ':not([type="image"]):not([type="file"]):not([disabled]),' +
      'select:not([disabled]),' +
      'textarea:not([disabled])'
    )

    for (const el of els) {
      if (!isVisible(el)) continue

      const type = (el.getAttribute('type') || el.tagName.toLowerCase()).toLowerCase()

      // Deduplicate radio groups — represent the whole group as one field
      if (type === 'radio') {
        const name = el.name
        if (!name || seenRadioGroups.has(name)) continue
        seenRadioGroups.add(name)
        fields.push({
          selector: `input[type="radio"][name="${CSS.escape(name)}"]`,
          type: 'radio',
          label: getFieldLabel(el),
          options: getRadioGroupOptions(name),
        })
        continue
      }

      const selector = getFieldSelector(el)
      if (seen.has(selector)) continue
      seen.add(selector)

      const field = { selector, type, label: getFieldLabel(el) }

      if (el.tagName === 'SELECT') {
        field.options = Array.from(el.options)
          .map(o => o.text.trim())
          .filter(t => t && t !== '—' && t !== 'Select...' && t !== 'Please select')
      }

      fields.push(field)
    }

    return fields
  }

  // ─── Field filler ─────────────────────────────────────────────────────────

  function fillFormFields(mappings) {
    let filled = 0

    for (const { selector, value } of mappings) {
      if (!value) continue
      try {
        const el = document.querySelector(selector)
        if (!el) continue

        const tag = el.tagName
        const type = (el.getAttribute('type') || '').toLowerCase()

        if (tag === 'SELECT') {
          const nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLSelectElement.prototype, 'value').set
          const options = Array.from(el.options)
          const match = options.find(o => o.text.trim() === value) ||
                        options.find(o => o.text.trim().toLowerCase() === value.toLowerCase())
          if (!match) continue
          nativeSetter.call(el, match.value)
          el.dispatchEvent(new Event('change', { bubbles: true }))
          filled++
        } else if (type === 'radio') {
          // selector is the group selector; find the matching option
          const radios = document.querySelectorAll(selector)
          for (const radio of radios) {
            const label = radio.id
              ? document.querySelector(`label[for="${CSS.escape(radio.id)}"]`)?.textContent?.trim()
              : radio.closest('label')?.textContent?.trim()
            const text = label || radio.value
            if (text.toLowerCase() === value.toLowerCase() || radio.value.toLowerCase() === value.toLowerCase()) {
              radio.click()
              filled++
              break
            }
          }
        } else if (type === 'checkbox') {
          const shouldCheck = value === 'true' || value === '1' || value.toLowerCase() === 'yes'
          if (el.checked !== shouldCheck) { el.click(); filled++ }
        } else {
          // text, email, tel, url, number, textarea
          const proto = tag === 'TEXTAREA' ? window.HTMLTextAreaElement.prototype : window.HTMLInputElement.prototype
          const nativeSetter = Object.getOwnPropertyDescriptor(proto, 'value').set
          nativeSetter.call(el, value)
          el.dispatchEvent(new Event('input', { bubbles: true }))
          el.dispatchEvent(new Event('change', { bubbles: true }))
          filled++
        }
      } catch (_) {}
    }

    return filled
  }
```

        ${msgHtml()}
        <button id="apply-btn" style="
          width:100%;padding:9px;background:#15803d;color:#fff;border:none;

  async function handleFill() {
    const fillBtn = shadow.getElementById('af-fill')
    if (fillBtn) { fillBtn.disabled = true; fillBtn.textContent = 'Scanning…'; fillBtn.style.opacity = '0.7' }

    // Merge edited simple fields into the profile sent to the API
    const effectiveProfile = { ...selectedProfile, ...editedFields }

    const formFields = scanFormFields()

    let coverLetterText = null

    if (coverLetterOn) {
      if (fillBtn) fillBtn.textContent = 'Generating cover letter…'
      const clRes = await chrome.runtime.sendMessage({
        type: 'COVER_LETTER',
        data: {
          profileId: selectedProfile.id,
          jobTitle: jobInfo.title || '',
          company: jobInfo.company || '',
          jobContext: filterPageForLLM(),
          customPrompt: selectedProfile.cover_letter_prompt || '',
        },
      }).catch(() => null)
      if (clRes?.cover_letter) coverLetterText = clRes.cover_letter
    }

    if (fillBtn) fillBtn.textContent = 'Filling…'

    const autofillRes = await chrome.runtime.sendMessage({
      type: 'AUTOFILL',
      data: {
        profileId: selectedProfile.id,
        formFields,
        jobTitle: jobInfo.title || '',
        company: jobInfo.company || '',
        profileOverrides: editedFields,   // inline edits from the review card
      },
    }).catch(() => null)

    if (!autofillRes?.fields) {
      if (fillBtn) { fillBtn.disabled = false; fillBtn.textContent = 'Fill form ✓'; fillBtn.style.opacity = '1' }
      showMsg('Autofill failed. Try again.', 'error')
      return
    }

    filledCount = fillFormFields(autofillRes.fields)

    coverLetterPasted = false
    if (coverLetterText) {
      // Find a visible textarea that likely is the cover letter field
      const textareas = Array.from(document.querySelectorAll('textarea:not([disabled])'))
        .filter(el => {
          const label = getFieldLabel(el).toLowerCase()
          return label.includes('cover') || label.includes('letter') || label.includes('why') || label.includes('motivation')
        })
      if (textareas.length > 0) {
        const proto = window.HTMLTextAreaElement.prototype
        const nativeSetter = Object.getOwnPropertyDescriptor(proto, 'value').set
        nativeSetter.call(textareas[0], coverLetterText)
        textareas[0].dispatchEvent(new Event('input', { bubbles: true }))
        textareas[0].dispatchEvent(new Event('change', { bubbles: true }))
        coverLetterPasted = true
        filledCount++
      }
    }

    autofillStep = 'done'
    render()
  }
